  (defvar bootstrap-version)
  (let ((bootstrap-file
         (expand-file-name
          "straight/repos/straight.el/bootstrap.el"
          (or (bound-and-true-p straight-base-dir)
              user-emacs-directory)))

        (goto-char (point-max))

  (use-package dired
    :straight (:type built-in)
    :config
    ;; 1. Prevent creating a new buffer for every directory you visit
    (setq dired-kill-when-opening-new-dired-buffer t)

    ;; 2. "Do What I Mean" target: If you have two Dired windows open,
    ;; copying/moving automatically suggests the other window as the target.
    (setq dired-dwim-target t)

    ;; 3. Sorting and styling (Requires 'ls' or 'gls' on macOS)
    ;; -h: human-readable sizes, --group-directories-first: folders on top
    (setq dired-listing-switches "-agho --group-directories-first --time-style=long-iso")

    ;; 4. Streamline operations and safety prompts
    (setq dired-recursive-copies 'always)      ; Don't ask when copying folders
    (setq dired-recursive-deletes 'top)        ; Ask once for top-level folder deletion
    (setq delete-by-moving-to-trash t)         ; Move to system trash instead of wiping
    (setq dired-clean-confirm-killing-deleted-buffers nil) ; Auto-close dead buffers

    ;; 5. Enable the "find-alternate-file" command (bound to 'a' by default)
    (put 'dired-find-alternate-file 'disabled nil)

    :hook
    ;; Hide ugly file details (permissions, links) by default. Toggle with '('
    (dired-mode . dired-hide-details-mode)
    ;; Auto-hide hidden/dot files by default. Toggle with '.'
    (dired-mode . dired-omit-mode)

    :bind
    ;; Quick shortcut to toggle hidden/dot files
    (:map dired-mode-map
          ("." . dired-omit-mode)))

#+BEGIN_SRC emacs-lisp
  (defun my/helpers/open-current-line-in-intellij ()
    "Open the current line in intellij."
    (interactive)
    (let* ((line-number (number-to-string (line-number-at-pos)))
           (file (buffer-file-name)))
      (call-process "idea" nil nil nil "--line" line-number file)))

#+BEGIN_SRC emacs-lisp
  (defun my/helpers/magit-status-ediff-current-file ()
    (interactive)
    (magit-section-case
      (t (let* (($current-file (magit-current-file)))
           (when $current-file
             (pcase (magit-diff-type)
               (`unstaged (magit-ediff-show-unstaged $current-file))
               (`staged (magit-ediff-show-staged $current-file))))))))
